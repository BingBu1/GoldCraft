cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(ParallelDownload LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "ParallelDownload requires Windows x86")
endif()
get_filename_component(WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(METAHOOK "${WORKSPACE}/external/MetaHookSv-20261007")
set(SDK "${METAHOOK}/MetaHook/include")
add_library(ParallelDownload SHARED plugin.cpp transfer.cpp "${SDK}/HLSDK/common/interface.cpp")
target_compile_features(ParallelDownload PRIVATE cxx_std_20)
target_include_directories(ParallelDownload PRIVATE "${SDK}" "${SDK}/Interface" "${SDK}/Interface/VGUI" "${SDK}/VGUI"
    "${SDK}/HLSDK/cl_dll" "${SDK}/HLSDK/common" "${SDK}/HLSDK/engine" "${SDK}/HLSDK/pm_shared" "${SDK}/HLSDK/public"
    "${SDK}/SourceSDK" "${SDK}/SourceSDK/tier0" "${SDK}/SourceSDK/tier1" "${METAHOOK}/MetaHook/thirdparty/capstone_fork/include"
    "${METAHOOK}/Plugins/VGUI2Extension/include/Interface")
target_compile_definitions(ParallelDownload PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS NO_MALLOC_OVERRIDE)
target_compile_options(ParallelDownload PRIVATE /utf-8 /W4 /Zi)
target_link_libraries(ParallelDownload PRIVATE winhttp)
target_link_options(ParallelDownload PRIVATE /DEBUG /INCREMENTAL:NO /OPT:REF /OPT:ICF)
add_executable(parallel_download_tests "${WORKSPACE}/tests/parallel-download/main.cpp" transfer.cpp)
target_compile_features(parallel_download_tests PRIVATE cxx_std_20)
target_include_directories(parallel_download_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_options(parallel_download_tests PRIVATE /utf-8 /W4)
target_link_libraries(parallel_download_tests PRIVATE winhttp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${WORKSPACE}/tests/parallel-download/generate_fs.py"
    "${SDK}/Interface/IFileSystem.h" "${CMAKE_CURRENT_BINARY_DIR}/filesystem_adapter.hpp" COMMAND_ERROR_IS_FATAL ANY)
add_executable(parallel_download_bridge_tests "${WORKSPACE}/tests/parallel-download/bridge.cpp" transfer.cpp "${SDK}/HLSDK/common/interface.cpp")
get_target_property(PLUGIN_INCLUDES ParallelDownload INCLUDE_DIRECTORIES)
get_target_property(PLUGIN_DEFINES ParallelDownload COMPILE_DEFINITIONS)
target_include_directories(parallel_download_bridge_tests PRIVATE ${PLUGIN_INCLUDES} "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(parallel_download_bridge_tests PRIVATE ${PLUGIN_DEFINES})
target_compile_features(parallel_download_bridge_tests PRIVATE cxx_std_20)
target_compile_options(parallel_download_bridge_tests PRIVATE /utf-8 /W4)
target_link_libraries(parallel_download_bridge_tests PRIVATE winhttp)
