cmake_minimum_required(VERSION 3.24)
project(goldcraft_http_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(http_download_tests http_download_tests.cpp
    ../http_download.cpp ../http_download_store.cpp ../http_download_server.cpp ../launcher_http.cpp)
target_include_directories(http_download_tests PRIVATE fixture .. ../../include)
target_link_libraries(http_download_tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(http_download_tests PRIVATE /utf-8 /UNDEBUG)
endif()
if(WIN32)
    target_link_libraries(http_download_tests PRIVATE ws2_32)
    add_test(NAME http_package COMMAND powershell -NoProfile -NonInteractive -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/Test-Package.ps1" -OutputDirectory "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(http_package PROPERTIES TIMEOUT 60)
endif()
add_test(NAME http_download COMMAND http_download_tests)
set_tests_properties(http_download PROPERTIES TIMEOUT 60)
add_executable(launcher_http_tests launcher_http_tests.cpp ../launcher_http.cpp ../http_download_server.cpp)
target_include_directories(launcher_http_tests PRIVATE ..)
target_link_libraries(launcher_http_tests PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(launcher_http_tests PRIVATE ws2_32)
endif()
if(MSVC)
    target_compile_options(launcher_http_tests PRIVATE /utf-8 /UNDEBUG)
endif()
add_test(NAME launcher_http COMMAND launcher_http_tests "${CMAKE_CURRENT_BINARY_DIR}/launcher-fixture")
set_tests_properties(launcher_http PROPERTIES TIMEOUT 30)
