cmake_minimum_required(VERSION 3.24)
project(csol_scenario_automatic_respawn_tests LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the project's approved Windows x86 Clang toolchain")
endif()
option(AUTOMATIC_RESPAWN_SDK "Execute frozen actual rules with controlled SDK boundaries" OFF)
if(AUTOMATIC_RESPAWN_SDK)
    add_executable(csol_scenario_automatic_respawn_tests sdk.cpp)
else()
    add_executable(csol_scenario_automatic_respawn_tests main.cpp)
endif()
target_compile_features(csol_scenario_automatic_respawn_tests PRIVATE cxx_std_20)
target_compile_options(csol_scenario_automatic_respawn_tests PRIVATE /utf-8 /W4 /fp:precise)
enable_testing()
add_test(NAME scenario_automatic_respawn COMMAND csol_scenario_automatic_respawn_tests)
set_tests_properties(scenario_automatic_respawn PROPERTIES TIMEOUT 20)
