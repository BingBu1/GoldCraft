cmake_minimum_required(VERSION 3.24)
project(CSOLScenarioGolden LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(DATA ../../native/csol/scenario)
add_executable(csol_scenario_golden main.cpp ${DATA}/monsters/golden.cpp ${DATA}/monsters/monster_types.cpp
    ${DATA}/core/config.cpp ${DATA}/core/spawn_points.cpp)
add_executable(csol_scenario_golden_runtime runtime_main.cpp ${DATA}/core/controller.cpp
    ${DATA}/core/waves.cpp ${DATA}/core/trigger_program.cpp)
find_path(RAPIDJSON_INCLUDE_DIR rapidjson/document.h PATHS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../external/MetaHookSv-20261007/MetaHook/thirdparty/rapidjson/include" REQUIRED)
target_include_directories(csol_scenario_golden PRIVATE "${RAPIDJSON_INCLUDE_DIR}")
foreach(target csol_scenario_golden csol_scenario_golden_runtime)
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /fp:precise /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    endif()
endforeach()
enable_testing()
add_test(NAME golden_parser COMMAND csol_scenario_golden --parser)
add_test(NAME golden_parameters COMMAND csol_scenario_golden --parameters)
add_test(NAME golden_candidates COMMAND csol_scenario_golden --candidates)
add_test(NAME golden_plans COMMAND csol_scenario_golden --plans)
add_test(NAME golden_members COMMAND csol_scenario_golden --members)
add_test(NAME golden_runtime COMMAND csol_scenario_golden_runtime)
